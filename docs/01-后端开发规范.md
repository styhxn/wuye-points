### 2.1 成功响应

```java
// ApiResponse.java
@Data
public class ApiResponse<T> {
    private int code;
    private String message;
    private T data;
    private String timestamp;
    private List<FieldError> errors; // 仅错误时有

    public static <T> ApiResponse<T> success(T data) {
        ApiResponse<T> response = new ApiResponse<>();
        response.setCode(0);
        response.setMessage("success");
        response.setData(data);
        response.setTimestamp(Instant.now().toString());
        return response;
    }

    public static <T> ApiResponse<T> success(String message, T data) {
        ApiResponse<T> response = new ApiResponse<>();
        response.setCode(0);
        response.setMessage(message);
        response.setData(data);
        response.setTimestamp(Instant.now().toString());
        return response;
    }

    public static <T> ApiResponse<T> error(int code, String message) {
        ApiResponse<T> response = new ApiResponse<>();
        response.setCode(code);
        response.setMessage(message);
        response.setData(null);
        response.setTimestamp(Instant.now().toString());
        return response;
    }
}
```

### 2.2 分页响应

```java
// PageResponse.java
@Data
public class PageResponse<T> {
    private List<T> list;
    private Pagination pagination;

    @Data
    public static class Pagination {
        private int page;
        private int pageSize;
        private long total;
        private int totalPages;
    }

    public static <T> PageResponse<T> of(List<T> list, int page, int pageSize, long total) {
        PageResponse<T> response = new PageResponse<>();
        response.setList(list);
        Pagination pagination = new Pagination();
        pagination.setPage(page);
        pagination.setPageSize(pageSize);
        pagination.setTotal(total);
        pagination.setTotalPages((int) Math.ceil((double) total / pageSize));
        response.setPagination(pagination);
        return response;
    }
}
```

```java
// BizException.java
@Getter
public class BizException extends RuntimeException {
    private final int code;

    public BizException(ErrorCode errorCode) {
        super(errorCode.getMessage());
        this.code = errorCode.getCode();
    }

    public BizException(int code, String message) {
        super(message);
        this.code = code;
    }
}

// GlobalExceptionHandler.java
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(BizException.class)
    public ResponseEntity<ApiResponse<Void>> handleBizException(BizException e) {
        log.warn("业务异常: code={}, message={}", e.getCode(), e.getMessage());
        return ResponseEntity
                .status(getHttpStatus(e.getCode()))
                .body(ApiResponse.error(e.getCode(), e.getMessage()));
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<Void>> handleValidation(MethodArgumentNotValidException e) {
        List<ApiResponse.FieldError> errors = e.getBindingResult().getFieldErrors().stream()
                .map(fe -> new ApiResponse.FieldError(fe.getField(), fe.getDefaultMessage(), fe.getRejectedValue()))
                .collect(Collectors.toList());
        ApiResponse<Void> response = ApiResponse.error(ErrorCode.PARAM_INVALID.getCode(), "参数校验失败", errors);
        return ResponseEntity.badRequest().body(response);
    }

    @ExceptionHandler(MethodArgumentTypeMismatchException.class)
    public ResponseEntity<ApiResponse<Void>> handleTypeMismatch(MethodArgumentTypeMismatchException e) {
        log.warn("参数类型错误: {}", e.getMessage());
        return ResponseEntity.badRequest()
                .body(ApiResponse.error(ErrorCode.PARAM_INVALID.getCode(), "参数类型错误: " + e.getName()));
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<ApiResponse<Void>> handleIllegalArgument(IllegalArgumentException e) {
        log.warn("非法参数: {}", e.getMessage());
        return ResponseEntity.badRequest()
                .body(ApiResponse.error(ErrorCode.PARAM_INVALID.getCode(), e.getMessage()));
    }

    @ExceptionHandler(OptimisticLockingFailureException.class)
    public ResponseEntity<ApiResponse<Void>> handleOptimisticLock(OptimisticLockingFailureException e) {
        log.warn("乐观锁冲突: {}", e.getMessage());
        return ResponseEntity.status(HttpStatus.CONFLICT)
                .body(ApiResponse.error(ErrorCode.ORDER_CONFLICT.getCode(), ErrorCode.ORDER_CONFLICT.getMessage()));
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiResponse<Void>> handleException(Exception e) {
        log.error("系统异常", e);
        return ResponseEntity
                .status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(ApiResponse.error(ErrorCode.SYSTEM_ERROR.getCode(), ErrorCode.SYSTEM_ERROR.getMessage()));
    }

    private HttpStatus getHttpStatus(int code) {
        if (code >= 20001 && code <= 20005) return HttpStatus.UNAUTHORIZED;
        if (code == 10001 || code == 10002 || code == 10003
                || code == 30003
                || code == 40001 || code == 50001 || code == 93004
                || code == 94004) return HttpStatus.BAD_REQUEST;
        if (code == 404 || code == 30001 || code == 60001 || code == 60004 || code == 70001 || code == 90001
                || code == 91001 || code == 91002 || code == 91003
                || code == 92001 || code == 92002 || code == 93001
                || code == 40002 || code == 90012 || code == 95001
                || code == 96001 || code == 96002 || code == 97001) {
            return HttpStatus.NOT_FOUND;
        }
        if (code == 20004 || code == 50002 || code == 93003) return HttpStatus.FORBIDDEN;
        if (code == 70003 || code == 60002 || code == 60003 || code == 70002 || code == 90002
                || code == 30002 || code == 30004 || code == 94001 || code == 94003
                || code == 90013 || code == 96003) return HttpStatus.CONFLICT;
        return HttpStatus.INTERNAL_SERVER_ERROR;
    }
}
```

## 5. JWT认证实现

```java
// JwtUtil.java
@Component
public class JwtUtil {
    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.access-token-expiration}")
    private long accessTokenExpiration;

    @Value("${jwt.refresh-token-expiration}")
    private long refreshTokenExpiration;

    public String generateAccessToken(Resident resident) {
        return Jwts.builder()
                .setSubject(resident.getId())
                .claim("role", resident.getRole())
                .claim("propertyCompanyId", resident.getPropertyCompanyId())
                .claim("familyId", resident.getFamilyId())
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + accessTokenExpiration))
                .signWith(SignatureAlgorithm.HS256, secret)
                .compact();
    }

    public String generateRefreshToken(Resident resident) {
        return Jwts.builder()
                .setSubject(resident.getId())
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + refreshTokenExpiration))
                .signWith(SignatureAlgorithm.HS256, secret)
                .compact();
    }

    public Claims parseToken(String token) {
        return Jwts.parser()
                .setSigningKey(secret)
                .parseClaimsJws(token)
                .getBody();
    }

    public boolean isTokenValid(String token) {
        try {
            Claims claims = parseToken(token);
            return !claims.getExpiration().before(new Date());
        } catch (Exception e) {
            return false;
        }
    }
}

// AuthInterceptor.java
@Component
public class AuthInterceptor implements HandlerInterceptor {
    @Autowired
    private JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        String authHeader = request.getHeader("Authorization");
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            throw new BizException(ErrorCode.TOKEN_MISSING);
        }

        String token = authHeader.substring(7);
        if (!jwtUtil.isTokenValid(token)) {
            throw new BizException(ErrorCode.TOKEN_EXPIRED);
        }

        Claims claims = jwtUtil.parseToken(token);
        request.setAttribute("userId", claims.getSubject());
        request.setAttribute("role", claims.get("role"));
        request.setAttribute("propertyCompanyId", claims.get("propertyCompanyId"));
        request.setAttribute("familyId", claims.get("familyId"));

        return true;
    }
}
```

## 6. ID生成器

```java
// SnowflakeIdGenerator.java
@Component
public class SnowflakeIdGenerator {
    private final long epoch = 1672531200000L; // 2023-01-01
    private final long machineIdBits = 5;
    private final long sequenceBits = 12;
    private final long maxMachineId = ~(-1L << machineIdBits);
    private final long maxSequence = ~(-1L << sequenceBits);

    private long machineId;
    private long sequence = 0L;
    private long lastTimestamp = -1L;

    @Value("${server.machine-id:1}")
    public void setMachineId(long machineId) {
        if (machineId > maxMachineId || machineId < 0) {
            throw new IllegalArgumentException("Machine ID out of range");
        }
        this.machineId = machineId;
    }

    public synchronized long nextId() {
        long timestamp = System.currentTimeMillis();
        if (timestamp < lastTimestamp) {
            throw new RuntimeException("Clock moved backwards");
        }
        if (timestamp == lastTimestamp) {
            sequence = (sequence + 1) & maxSequence;
            if (sequence == 0) {
                timestamp = waitNextMillis(lastTimestamp);
            }
        } else {
            sequence = 0L;
        }
        lastTimestamp = timestamp;
        return ((timestamp - epoch) << (machineIdBits + sequenceBits))
                | (machineId << sequenceBits)
                | sequence;
    }

    private long waitNextMillis(long lastTimestamp) {
        long timestamp = System.currentTimeMillis();
        while (timestamp <= lastTimestamp) {
            timestamp = System.currentTimeMillis();
        }
        return timestamp;
    }

    public String generateId(String prefix) {
        return prefix + "_" + Long.toHexString(nextId()).substring(0, 8);
    }
}
```

```java
        }
    }
```

## 8. 分成计算服务（四级分成）

> **重要**：PRD V4.1 采用四级分成机制，积分按**住户所属物业 T** 计算（非商家所属物业）。
> 详细公式参见 `07-业务规则详解.md`。

### 8.1 四级分成公式

```
消费金额 P × 商家在该物业的抽佣比例 R_T = 可分利润 D
（D 中再扣除积分成本、物业币成本后为实际可分金额 D'）

物业分成   = D' × propertyShareRate     (默认 30%)
统筹分成   = D' × coordinatorShareRate  (默认 70%)
板块负责人 = 统筹分成 × sectorLeaderRate (默认 10%，从统筹中抽取)
个体负责人 = (统筹分成 - 板块负责人) × individualLeaderRate (默认 0%，从板块剩余中抽取)
统筹实收   = 统筹分成 - 板块负责人 - 个体负责人
```

### 8.2 DistributionService 实现

```java
// DistributionService.java
@Service
@Slf4j
public class DistributionService {
    @Autowired
    private PropertyCompanyRepository propertyCompanyRepository;
    @Autowired
    private MerchantPropertyConfigRepository merchantPropertyConfigRepository;
    @Autowired
    private CoordinatorRepository coordinatorRepository;
    @Autowired
    private SectorLeaderRepository sectorLeaderRepository;
    @Autowired
    private IndividualLeaderRepository individualLeaderRepository;
    @Autowired
    private DistributionRecordRepository distributionRecordRepository;

    /**
     * 计算并持久化四级分成
     * 注意：propertyCompanyId 取自 resident（住户所属物业 T），而非 merchant。
     *
     * @param order 订单（含消费金额 P）
     * @param residentId 住户ID（用于确定所属物业 T）
     * @return 分成结果
     */
    @Transactional(rollbackFor = Exception.class)
    public DistributionResult calculate(Order order, String residentId) {
        Resident resident = residentRepository.findById(residentId)
                .orElseThrow(() -> new BizException(ErrorCode.RESIDENT_NOT_FOUND));

        // 关键：使用住户所属物业 T 计算分成
        String propertyCompanyId = resident.getPropertyCompanyId();
        PropertyCompany pc = propertyCompanyRepository.findById(propertyCompanyId)
                .orElseThrow(() -> new BizException(ErrorCode.SYSTEM_ERROR));

        // 商家在该物业下的抽佣比例（来自 merchant_property_configs）
        MerchantPropertyConfig mpc = merchantPropertyConfigRepository
                .findByMerchantIdAndPropertyCompanyId(order.getMerchantId(), propertyCompanyId)
                .orElseThrow(() -> new BizException(ErrorCode.MERCHANT_NOT_FOUND));

        BigDecimal P = order.getTotalAmount();
        BigDecimal R_T = mpc.getCommissionRate();

        // 扣除积分成本与物业币成本
        BigDecimal pointCost = order.getPointCost();     // P × 兑换比
        BigDecimal coinCost = order.getCoinCost();       // 物业币返现成本
        BigDecimal distributableAmount = P.multiply(R_T)
                .subtract(pointCost).subtract(coinCost)
                .max(BigDecimal.ZERO);

        // 物业公司 30%
        BigDecimal propertyShare = distributableAmount.multiply(pc.getPropertyShareRate());
        // 统筹 70%
        BigDecimal coordinatorShare = distributableAmount.multiply(pc.getCoordinatorShareRate());
        // 板块负责人（从统筹中抽取，默认 10%）
        BigDecimal sectorLeaderShare = coordinatorShare.multiply(pc.getSectorLeaderRate());
        // 个体负责人（从板块剩余中抽取，默认 0%）
        BigDecimal individualLeaderShare = sectorLeaderShare.multiply(pc.getIndividualLeaderRate());
        // 统筹实收
        BigDecimal coordinatorActual = coordinatorShare
                .subtract(sectorLeaderShare)
                .add(individualLeaderShare); // 个体从板块中抽走，剩余归统筹

        // 定位统筹 / 板块 / 个体 负责人
        Coordinator coordinator = coordinatorRepository
                .findByPropertyCompanyIdAndSector(propertyCompanyId, order.getSector())
                .orElse(null);
        SectorLeader sectorLeader = (coordinator == null) ? null :
                sectorLeaderRepository.findByCoordinatorIdAndSector(coordinator.getId(), order.getSector())
                        .orElse(null);
        IndividualLeader individualLeader = (sectorLeader == null) ? null :
                individualLeaderRepository.findBySectorLeaderIdAndMerchantId(
                        sectorLeader.getId(), order.getMerchantId()).orElse(null);

        // 持久化分成记录
        DistributionRecord record = DistributionRecord.builder()
                .id(idGenerator.generateId("dr_"))
                .orderId(order.getId())
                .merchantId(order.getMerchantId())
                .propertyCompanyId(propertyCompanyId)
                .residentId(residentId)
                .totalAmount(P)
                .commissionRate(R_T)
                .pointCost(pointCost)
                .coinCost(coinCost)
                .distributableAmount(distributableAmount)
                .propertyShare(propertyShare)
                .coordinatorShare(coordinatorActual)
                .sectorLeaderShare(sectorLeaderShare)
                .individualLeaderShare(individualLeaderShare)
                .coordinatorId(coordinator == null ? null : coordinator.getId())
                .sectorLeaderId(sectorLeader == null ? null : sectorLeader.getId())
                .individualLeaderId(individualLeader == null ? null : individualLeader.getId())
                .build();
        distributionRecordRepository.save(record);

        return DistributionResult.builder()
                .totalAmount(P)
                .commissionRate(R_T)
                .distributableAmount(distributableAmount)
                .propertyShare(propertyShare)
                .coordinatorShare(coordinatorActual)
                .sectorLeaderShare(sectorLeaderShare)
                .individualLeaderShare(individualLeaderShare)
                .build();
    }
}

// DistributionResult.java
@Data
@Builder
public class DistributionResult {
    private BigDecimal totalAmount;
    private BigDecimal commissionRate;
    private BigDecimal distributableAmount;
    private BigDecimal propertyShare;       // 物业公司收入
    private BigDecimal coordinatorShare;    // 统筹实收
    private BigDecimal sectorLeaderShare;   // 板块负责人收入
    private BigDecimal individualLeaderShare; // 个体负责人收入
}
```

### 8.3 数据隔离机制（强制）

多物业公司架构下，**所有业务表的查询必须强制过滤 `property_company_id`**，禁止跨物业数据访问：

```java

    Page<Merchant> findActiveByPropertyCompany(@Param("pcId") String pcId, Pageable pageable);

// ❌ 错误：未过滤 property_company_id，存在跨物业数据泄露风险

// 在 Service 层从 SecurityContext 获取当前用户的 propertyCompanyId
@RequiredArgsConstructor
public abstract class BaseService {
    protected String currentPropertyCompanyId() {
        return SecurityContextHolder.getContext().getAuthentication()
                .getDetails().toString(); // 从 JWT 注入
    }
```

- **JWT 携带**：登录时将 `propertyCompanyId`、`role`、`userId` 写入 JWT
- **拦截器注入**：`AuthInterceptor` 解析 JWT 后将 `propertyCompanyId` 放入 `SecurityContext`
- **Repository 强制**：所有查询方法必须显式带 `property_company_id` 条件
- **跨物业操作**：仅 `platform_admin` 角色允许，需 `@RequireRole("platform_admin")` 注解
